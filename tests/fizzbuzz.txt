? If the number is a multiple of 5, say Buzz
? If the number is a multiple of both, say FizzBuzz
? If the number is none of these, say the number

    define integer i as 1
    loop count
        define integer modThree as i % 3
        define integer modFive as i % 5

        if modThree = 0

            print "FizzBuzz"
        endif 
        endif

            print "Fizz"
        endif

        if modFive = 0
            print "Buzz"
        endif

        if modThree != 0
        if modFive != 0
            print i
        endif
        endif
